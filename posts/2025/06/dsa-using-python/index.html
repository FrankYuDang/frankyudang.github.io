  4. 基于list 的栈结构声明

    class Stack(list):
    def __init__(self):
        super().__init__()
    def push(self,item):
        self.append(item)
    def pop(self):
        return super().pop()
    def peek(self): 
        return self[-1]
    def size(self):
        return len(self)
    def __repr__(self):
        l = len(self) * 7
        s = "|" + "-"*l +")\n|"
        for a in self:
            s += "| %-5s" % a 
        s += "\n|" + "-"*l +")"
        return s 

    __str__ = __repr__ 


if __name__ == "__main__":
    st = Stack() 
    st.push("R")
    st.push("win")
    print(st)
    st.pop()
    print(st.peek())
